import uuid
from sqlalchemy.types import TypeDecorator, CHAR
from sqlalchemy.dialects.postgresql import UUID

class GUID(TypeDecorator):
    """Platform-independent GUID type.
    Uses PostgreSQL's UUID type, otherwise uses CHAR(36)"""
    impl = CHAR
    cache_ok = True

    def load_dialect_impl(self, dialect):
        if dialect.name == 'postgresql':
            return dialect.type_descriptor(UUID())
        else:
            return dialect.type_descriptor(CHAR(36))

    def process_bind_param(self, value, dialect):
        if value is None:
            return value
        if dialect.name == 'postgresql':
            return str(value)
        if not isinstance(value, uuid.UUID):
            return str(uuid.UUID(value))
        return str(value)

    def process_result_value(self, value, dialect):
        if value is None:
            return value
        return str(value)

def admin_required(f):
    @wraps(f)
    @jwt_required()
    def decorated(*args, **kwargs):
        user_id = get_jwt_identity()
        user = User.query.get(int(user_id))
        if not user or not (user.has_role('super_admin') or user.has_role('faz_admin')):
            return jsonify({'error': 'Admin access required'}), 403
        return f(*args, **kwargs)
    return decorated

@bp.route('/matches/<int:match_id>', methods=['PUT'])
@admin_required
def update_match(match_id):
    match = Match.query.get_or_404(match_id)
    data = request.get_json()
    
    # Update basic match fields
    updatable = ['home_team', 'away_team', 'venue', 'date_time', 'description', 'is_active']
    for field in updatable:
        if field in data:
            setattr(match, field, data[field])
    
    # Handle ticket types
    if 'ticket_types' in data:
        incoming = []
        for tt_data in data['ticket_types']:
            name = tt_data.get('name', '').strip()
            price = tt_data.get('price')
            capacity = tt_data.get('capacity')
            if name and price is not None and capacity is not None:
                incoming.append({
                    'id': tt_data.get('id'),
                    'name': name,
                    'price': float(price),
                    'capacity': int(capacity)
                })
        
        # Get existing ticket types
        existing_map = {tt.id: tt for tt in match.ticket_types}
        
        # Track which existing IDs are kept
        kept_ids = set()
        
        for item in incoming:
            if item['id'] and item['id'] in existing_map:
                # Update existing
                tt = existing_map[item['id']]
                tt.name = item['name']
                tt.price = item['price']
                tt.capacity = item['capacity']
                kept_ids.add(item['id'])
            else:
                # Create new ticket type
                new_tt = TicketType(
                    match_id=match.id,
                    name=item['name'],
                    price=item['price'],
                    capacity=item['capacity']
                )
                db.session.add(new_tt)
                db.session.flush()  # Force assignment of ID
                kept_ids.add(new_tt.id)
        
        # Delete ticket types not in kept_ids
        for tt in match.ticket_types:
            if tt.id not in kept_ids:
                db.session.delete(tt)
    
    try:
        db.session.commit()
        print("Commit successful")
    except Exception as e:
        db.session.rollback()
        print(f"Commit failed: {e}")
        return jsonify({'error': str(e)}), 500
    
    # Refresh match to get updated ticket types
    db.session.refresh(match)
    return jsonify(match.to_dict()), 200

from flask import jsonify, request
from app.api import bp
from app import db
from app.models.hold import SeatHold
from app.models.ticket import TicketType
from datetime import datetime, timedelta

def clean_expired_holds():
    """Delete expired holds."""
    SeatHold.query.filter(SeatHold.expires_at < datetime.utcnow()).delete()
    db.session.commit()

@bp.route('/matches/<int:match_id>/hold', methods=['POST'])
def hold_tickets(match_id):
    data = request.get_json()
    if not data:
        return jsonify({'error': 'No data provided'}), 400

    ticket_type_id = data.get('ticket_type_id')
    try:
        quantity = int(data.get('quantity', 0))
    except (ValueError, TypeError):
        return jsonify({'error': 'Invalid quantity'}), 400

    hold_token = data.get('hold_token')

    clean_expired_holds()

    ticket_type = TicketType.query.get(ticket_type_id)
    if not ticket_type or ticket_type.match_id != match_id:
        return jsonify({'error': 'Ticket type not found'}), 404

    current_hold = None
    if hold_token:
        current_hold = SeatHold.query.filter_by(hold_token=hold_token).first()
        if not current_hold:
            return jsonify({'error': 'Hold token not found'}), 404

    held_by_others = db.session.query(db.func.sum(SeatHold.quantity)).filter(
        SeatHold.ticket_type_id == ticket_type_id,
        SeatHold.id != (current_hold.id if current_hold else 0)
    ).scalar() or 0

    available = ticket_type.capacity - ticket_type.tickets_sold - held_by_others
    if quantity > available:
        return jsonify({'error': f'Only {available} seats available', 'available': available}), 409

    if current_hold:
        current_hold.quantity = quantity
        current_hold.expires_at = datetime.utcnow() + timedelta(minutes=10)
        db.session.commit()
        return jsonify({
            'hold_token': current_hold.hold_token,
            'quantity': quantity,
            'available': available - quantity
        }), 200
    else:
        if quantity == 0:
            return jsonify({'error': 'Quantity must be positive'}), 400
        new_hold = SeatHold(
            match_id=match_id,
            ticket_type_id=ticket_type_id,
            quantity=quantity
        )
        db.session.add(new_hold)
        db.session.commit()
        return jsonify({
            'hold_token': new_hold.hold_token,
            'quantity': quantity,
            'available': available - quantity
        }), 201

@bp.route('/matches/<int:match_id>/hold/<hold_token>', methods=['DELETE'])
def release_hold(match_id, hold_token):
    hold = SeatHold.query.filter_by(hold_token=hold_token, match_id=match_id).first()
    if not hold:
        return jsonify({'error': 'Hold not found'}), 404
    db.session.delete(hold)
    db.session.commit()
    return jsonify({'message': 'Hold released'}), 200

from flask import jsonify, request
from flask_jwt_extended import jwt_required, get_jwt_identity, decode_token
from app.api import bp
from app import db
from app.models.order import Order, OrderItem
from app.models.ticket import TicketType
from app.models.hold import SeatHold
import uuid

@bp.route('/orders', methods=['POST'])
def create_order():
    try:
        data = request.get_json()
        
        # Get user_id if token is present (optional)
        user_id = None
        auth_header = request.headers.get('Authorization')
        if auth_header and auth_header.startswith('Bearer '):
            try:
                token = auth_header.split(' ')[1]
                decoded = decode_token(token)
                user_id = int(decoded['sub'])
            except:
                pass  # Guest checkout
        
        customer_email = data.get('customer_email')
        customer_name = data.get('customer_name')
        customer_phone = data.get('customer_phone')
        items = data.get('items', [])
        hold_tokens = data.get('hold_tokens', [])   # list of hold tokens to release

        if not customer_email or not items:
            return jsonify({'error': 'Email and items required'}), 400

        with db.session.begin_nested():
            total = 0
            order_items = []
            for item in items:
                ticket_type_id = item.get('ticket_type_id')
                quantity = item.get('quantity')
                if not ticket_type_id or not quantity or quantity <= 0:
                    db.session.rollback()
                    return jsonify({'error': 'Invalid item data'}), 400

                # Lock the row to prevent over‑selling
                ticket_type = db.session.query(TicketType).filter(
                    TicketType.id == ticket_type_id
                ).with_for_update().one_or_none()
                if not ticket_type:
                    db.session.rollback()
                    return jsonify({'error': f'Ticket type {ticket_type_id} not found'}), 404

                available = ticket_type.capacity - ticket_type.tickets_sold
                if quantity > available:
                    db.session.rollback()
                    return jsonify({'error': f'Only {available} tickets available for {ticket_type.name}'}), 400

                total += float(ticket_type.price) * quantity
                order_items.append({
                    'ticket_type': ticket_type,
                    'quantity': quantity,
                    'price': float(ticket_type.price)
                })

                # Reserve seats immediately
                ticket_type.tickets_sold += quantity
                db.session.add(ticket_type)

            # Create order (pending payment)
            order = Order(
                user_id=user_id,
                customer_email=customer_email,
                customer_name=customer_name,
                customer_phone=customer_phone,
                total_amount=total,
                payment_status='pending',
                payment_method='demo'
            )
            db.session.add(order)
            db.session.flush()

            # Create order items
            for item in order_items:
                order_item = OrderItem(
                    order_id=order.id,
                    ticket_type_id=item['ticket_type'].id,
                    quantity=item['quantity'],
                    unit_price=item['ticket_type'].price
                )
                db.session.add(order_item)

            # Delete the hold tokens if provided (they are no longer needed)
            for token in hold_tokens:
                hold = SeatHold.query.filter_by(hold_token=token).first()
                if hold:
                    db.session.delete(hold)

        db.session.commit()
        return jsonify(order.to_dict()), 201

    except Exception as e:
        db.session.rollback()
        print(f"Error creating order: {str(e)}")
        import traceback
        traceback.print_exc()
        return jsonify({'error': str(e)}), 500

    order = Order.query.get_or_404(order_id)
    if order.payment_status != 'pending':
        return jsonify({'error': 'Only pending orders can be cancelled'}), 400

    # Release reserved seats
    for item in order.items:
        ticket_type = TicketType.query.get(item.ticket_type_id)
        if ticket_type:
            ticket_type.tickets_sold = max(0, ticket_type.tickets_sold - item.quantity)
            db.session.add(ticket_type)

    db.session.delete(order)
    db.session.commit()
    return jsonify({'message': 'Order cancelled'}), 200

def club_access_required(f):
    """Allow club officers, admins, and handle missing club gracefully"""
    @wraps(f)
    @jwt_required()
    def decorated(*args, **kwargs):
        user_id = get_jwt_identity()
        user = User.query.get(int(user_id))
        if not user:
            return jsonify({'error': 'User not found'}), 401
        
        # Admins always pass
        if user.has_role('super_admin') or user.has_role('faz_admin'):
            return f(*args, **kwargs)
        
        # Club officers pass (even without club_id - route will handle)
        if user.has_role('club_admin') or user.has_role('club_officer'):
            return f(*args, **kwargs)
        
        return jsonify({'error': 'Access denied'}), 403
    return decorated

class Match(db.Model):
    __tablename__ = 'matches'
    id = db.Column(GUID(), primary_key=True, default=lambda: str(uuid.uuid4()))
    home_team = db.Column(db.String(100), nullable=False)
    away_team = db.Column(db.String(100), nullable=False)
    match_type = db.Column(db.String(20), nullable=False)
    club_id = db.Column(GUID(), db.ForeignKey('clubs.id'), nullable=True)
    venue = db.Column(db.String(200), nullable=False)
    date_time = db.Column(db.DateTime, nullable=False)
    description = db.Column(db.Text)
    is_active = db.Column(db.Boolean, default=True)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    created_by = db.Column(GUID(), db.ForeignKey('users.id'))

    club = db.relationship('Club', back_populates='matches')
    ticket_types = db.relationship('TicketType', back_populates='match', lazy='dynamic', cascade='all, delete-orphan')
    tickets = db.relationship('Ticket', back_populates='match', lazy='dynamic')

    def to_dict(self):
        from app.models.hold import SeatHold

        total_capacity = sum(tt.capacity for tt in self.ticket_types)
        total_sold = sum(tt.tickets_sold for tt in self.ticket_types)

        holds = db.session.query(
            SeatHold.ticket_type_id,
            func.sum(SeatHold.quantity).label('held')
        ).filter(
            SeatHold.match_id == self.id,
            SeatHold.expires_at > datetime.utcnow()
        ).group_by(SeatHold.ticket_type_id).all()
        held_map = {h.ticket_type_id: h.held for h in holds}

        for tt in self.ticket_types:
            tt.held = held_map.get(tt.id, 0)
            tt.dynamic_available = tt.capacity - tt.tickets_sold - tt.held

        return {
            'id': self.id,
            'home_team': self.home_team,
            'away_team': self.away_team,
            'match_type': self.match_type,
            'club_id': self.club_id,
            'venue': self.venue,
            'date_time': self.date_time.isoformat() if self.date_time else None,
            'description': self.description,
            'is_active': self.is_active,
            'ticket_types': [tt.to_dict() for tt in self.ticket_types],
            'total_capacity': total_capacity,
            'total_tickets_sold': total_sold,
            'total_available': total_capacity - total_sold - sum(held_map.values())
        }

@bp.route('/deposit', methods=['POST'])
def initiate_deposit():
    """
    Initiate a mobile money payment via PawaPay.
    Expects JSON: { order_id, provider, phone_number, amount, customer_email }
    Returns depositId on success (when prompt is sent).
    """
    data = request.get_json()
    if not data:
        return jsonify({'error': 'No data provided'}), 400

    order_id = data.get('order_id')
    provider = data.get('provider')
    phone_number = data.get('phone_number')
    amount = data.get('amount')
    customer_email = data.get('customer_email')

    if not all([order_id, provider, phone_number, amount]):
        return jsonify({'error': 'Missing required parameters'}), 400

    order = Order.query.get(order_id)
    if not order:
        return jsonify({'error': 'Order not found'}), 404

    if order.payment_status != 'pending':
        return jsonify({'error': 'Order is already paid or cancelled'}), 400

    deposit_id = str(uuid.uuid4())
    client_ref = f"FAZ-{order.id}-{uuid.uuid4().hex[:6].upper()}"

    payload = {
        "depositId": deposit_id,
        "payer": {
            "type": "MMO",
            "accountDetails": {
                "phoneNumber": phone_number,
                "provider": provider
            }
        },
        "amount": str(amount),
        "currency": "ZMW",
        "clientReferenceId": client_ref,
        "customerMessage": f"FAZ Order {order.id}"[:22],
        "metadata": [
            {"orderId": str(order.id)},
            {"customerId": customer_email or order.customer_email, "isPII": True}
        ]
    }

    try:
        resp = requests.post(
            f"{PAWAPAY_BASE_URL}/deposits",
            json=payload,
            headers=get_pawapay_headers(),
            timeout=30
        )
        pawa_resp = resp.json()

        if resp.status_code == 200:
            status = pawa_resp.get('status')
            if status in ['ACCEPTED', 'DUPLICATE_IGNORED']:
                # Save deposit ID to order for later polling
                order.payment_reference = pawa_resp.get('depositId', deposit_id)
                order.payment_method = provider
                db.session.commit()
                return jsonify({
                    'success': True,
                    'depositId': order.payment_reference,
                    'message': 'Payment request sent. Please check your phone to approve.'
                }), 200
            else:
                # Immediately rejected (e.g., invalid phone, provider unavailable)
                return jsonify({
                    'success': False,
                    'status': status,
                    'failureReason': pawa_resp.get('failureReason')
                }), 200
        else:
            return jsonify({
                'success': False,
                'status': pawa_resp.get('status', 'ERROR'),
                'failureReason': pawa_resp.get('failureReason'),
                'http_status': resp.status_code
            }), 200 if resp.status_code in [400, 401, 403] else 502

    except requests.exceptions.RequestException:
        return jsonify({'error': 'Payment service unavailable'}), 503

@bp.route('/deposit/<deposit_id>/check', methods=['GET'])
def check_deposit_status(deposit_id):
    """
    Poll PawaPay for the deposit status.
    If COMPLETED, validate seat availability and generate tickets.
    If FAILED or NOT_FOUND, mark the order as failed.
    """
    # Find the order by deposit ID stored in payment_reference
    order = Order.query.filter_by(payment_reference=deposit_id).first()
    if not order:
        return jsonify({'error': 'Deposit not found'}), 404

    try:
        resp = requests.get(
            f"{PAWAPAY_BASE_URL}/deposits/{deposit_id}",
            headers=get_pawapay_headers(),
            timeout=10
        )
        data = resp.json()

        if resp.status_code == 200:
            status = data.get('status')
            deposit_data = data.get('data') if status == 'FOUND' else None

            if status == 'FOUND' and deposit_data:
                deposit_status = deposit_data.get('status')

                if deposit_status == 'COMPLETED':
                    # Double‑check that all ordered ticket types still have enough capacity
                    for item in order.items:
                        ticket_type = TicketType.query.get(item.ticket_type_id)
                        if not ticket_type:
                            order.payment_status = 'failed'
                            db.session.commit()
                            return jsonify({
                                'success': False,
                                'status': 'FAILED',
                                'failureReason': {'failureMessage': 'Ticket type no longer exists'}
                            }), 200
                        available = ticket_type.capacity - ticket_type.tickets_sold
                        if item.quantity > available:
                            # Not enough seats left – fail the order
                            order.payment_status = 'failed'
                            db.session.commit()
                            return jsonify({
                                'success': False,
                                'status': 'FAILED',
                                'failureReason': {
                                    'failureMessage': f'Only {available} tickets available for {ticket_type.name}'
                                }
                            }), 200

                    # All items are available – generate tickets and update sold counts
                    for item in order.items:
                        ticket_type = TicketType.query.get(item.ticket_type_id)
                        for _ in range(item.quantity):
                            ticket = Ticket(
                                ticket_number=str(uuid.uuid4()),
                                order_item_id=item.id,
                                match_id=ticket_type.match_id,
                                ticket_type_id=ticket_type.id
                            )
                            ticket.generate_qr_code()
                            db.session.add(ticket)
                        ticket_type.tickets_sold += item.quantity
                        db.session.add(ticket_type)

                    order.payment_status = 'paid'
                    db.session.commit()
                    return jsonify({
                        'success': True,
                        'status': 'COMPLETED',
                        'message': 'Payment completed. Tickets generated.'
                    }), 200

                elif deposit_status == 'FAILED':
                    order.payment_status = 'failed'
                    db.session.commit()
                    return jsonify({
                        'success': False,
                        'status': 'FAILED',
                        'failureReason': deposit_data.get('failureReason')
                    }), 200

                elif deposit_status in ['ACCEPTED', 'PROCESSING', 'IN_RECONCILIATION']:
                    return jsonify({
                        'success': True,   # still pending, not a failure
                        'status': deposit_status,
                        'message': 'Payment still pending'
                    }), 200

                else:
                    return jsonify({'success': False, 'status': 'UNKNOWN'}), 200

            elif status == 'NOT_FOUND':
                order.payment_status = 'failed'
                db.session.commit()
                return jsonify({'success': False, 'status': 'NOT_FOUND'}), 200

            else:
                return jsonify({'success': False, 'status': 'UNKNOWN'}), 200

        else:
            return jsonify({
                'success': False,
                'status': 'ERROR',
                'failureReason': data.get('failureReason')
            }), 502

    except requests.exceptions.RequestException:
        return jsonify({'error': 'Payment service unavailable'}), 503

import uuid
from sqlalchemy.types import TypeDecorator, CHAR
from sqlalchemy.dialects.postgresql import UUID

class GUID(TypeDecorator):
    """Platform-independent GUID type."""
    impl = CHAR
    cache_ok = True

    def load_dialect_impl(self, dialect):
        if dialect.name == 'postgresql':
            return dialect.type_descriptor(UUID())
        else:
            return dialect.type_descriptor(CHAR(36))

    def process_bind_param(self, value, dialect):
        if value is None:
            return value
        if dialect.name == 'postgresql':
            return str(value)
        if not isinstance(value, uuid.UUID):
            return str(uuid.UUID(value))
        return str(value)

    def process_result_value(self, value, dialect):
        if value is None:
            return value
        return str(value)

def run_migrations_offline():
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url, target_metadata=get_metadata(), literal_binds=True
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online():
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """

    # this callback is used to prevent an auto-migration from being generated
    # when there are no changes to the schema
    # reference: http://alembic.zzzcomputing.com/en/latest/cookbook.html
    def process_revision_directives(context, revision, directives):
        if getattr(config.cmd_opts, 'autogenerate', False):
            script = directives[0]
            if script.upgrade_ops.is_empty():
                directives[:] = []
                logger.info('No changes in schema detected.')

    conf_args = current_app.extensions['migrate'].configure_args
    if conf_args.get("process_revision_directives") is None:
        conf_args["process_revision_directives"] = process_revision_directives

        context.configure(
            connection=connection,
            target_metadata=get_metadata(),
            **conf_args
        )

        with context.begin_transaction():
            context.run_migrations()

import { useState, useMemo, useRef } from 'react'
import ReactPaginate from 'react-paginate'
import Papa from 'papaparse'
import { Button, Input, Select, Card } from './ui'
import { ChevronLeft, ChevronRight, Grid, List, Download, Printer, Search, X } from 'lucide-react'

export default function DataTable({
  columns,
  data,
  loading,
  title,
  actions,
  searchable = true,
  exportable = true,
  pageSizeOptions = [5, 10, 20, 50],
  defaultPageSize = 5,
  gridView = false,
  onRowClick,
  showSerialNumber = true,
  emptyState = {
    icon: '📭',
    title: 'No data found',
    description: 'There are no items to display.'
  }
}) {
  const [searchTerm, setSearchTerm] = useState('')
  const [currentPage, setCurrentPage] = useState(0)
  const [pageSize, setPageSize] = useState(defaultPageSize)
  const [viewMode, setViewMode] = useState(gridView ? 'grid' : 'table')
  const [columnFilters, setColumnFilters] = useState({})
  const tableRef = useRef(null)

  // Filter data based on search and column filters
  const filteredData = useMemo(() => {
    return data.filter(item => {
      // Global search
      if (searchTerm) {
        const matchesSearch = columns.some(col => {
          const value = col.accessor ? col.accessor(item) : item[col.key]
          return String(value).toLowerCase().includes(searchTerm.toLowerCase())
        })
        if (!matchesSearch) return false
      }

      // Column filters
      for (const [key, filterValue] of Object.entries(columnFilters)) {
        if (!filterValue) continue
        const col = columns.find(c => c.key === key)
        const value = col?.accessor ? col.accessor(item) : item[key]
        if (!String(value).toLowerCase().includes(filterValue.toLowerCase())) return false
      }

      return true
    })
  }, [data, searchTerm, columnFilters, columns])

  // Pagination
  const pageCount = Math.ceil(filteredData.length / pageSize)
  const paginatedData = filteredData.slice(
    currentPage * pageSize,
    (currentPage + 1) * pageSize
  )

  const handlePageChange = ({ selected }) => setCurrentPage(selected)

  // Export: only visible columns (no actions)
  const exportColumns = showSerialNumber 
    ? [{ key: 'sn', header: 'S/N', accessor: (_, idx) => idx + 1 }, ...columns]
    : columns

  const getExportData = () => {
    return filteredData.map((item, index) => {
      const row = {}
      exportColumns.forEach(col => {
        const value = col.accessor 
          ? col.accessor(item, index + (currentPage * pageSize))
          : item[col.key]
        row[col.header] = value
      })
      return row
    })
  }

  const handleExportCSV = () => {
    const exportData = getExportData()
    const csv = Papa.unparse(exportData)
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
    const link = document.createElement('a')
    link.href = URL.createObjectURL(blob)
    link.download = `${title || 'export'}.csv`
    link.click()
  }

  const handlePrint = () => {
    const printWindow = window.open('', '_blank')
    const tableHtml = tableRef.current?.cloneNode(true)
    // Remove actions column from print
    if (tableHtml) {
      const rows = tableHtml.querySelectorAll('tr')
      rows.forEach(row => {
        const lastCell = row.lastElementChild
        if (lastCell && lastCell.textContent.includes('Actions')) {
          lastCell.remove()
        } else if (row.cells.length > exportColumns.length) {
          // Remove extra cells (actions)
          for (let i = exportColumns.length; i < row.cells.length; i++) {
            row.deleteCell(-1)
          }
        }
      })
    }
    printWindow.document.write(`
      <html>
        <head><title>${title || 'Print'}</title>
        <style>
          body { font-family: sans-serif; margin: 20px; }
          table { border-collapse: collapse; width: 100%; }
          th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
          th { background: #f5f5f5; }
        </style>
        </head>
        <body>${tableHtml?.outerHTML || ''}</body>
      </html>
    `)
    printWindow.document.close()
    printWindow.print()
  }

  const handleExportPDF = () => {
    // Use browser print to save as PDF
    handlePrint()
  }

  const clearAllFilters = () => {
    setSearchTerm('')
    setColumnFilters({})
  }

  const hasActiveFilters = searchTerm || Object.values(columnFilters).some(v => v)

  // Add serial number to columns if enabled
  const displayColumns = showSerialNumber 
    ? [{ key: 'sn', header: 'S/N', render: (_, idx) => <span className="text-gray-500 text-sm">{idx + 1 + (currentPage * pageSize)}</span> }, ...columns]
    : columns

  if (loading) {
    return (
      <div className="flex justify-center py-20">
        <div className="w-10 h-10 border-4 border-green-600 border-t-transparent rounded-full animate-spin" />
      </div>
    )
  }

  return (
    <div className="space-y-4">
      {/* Toolbar */}
      <div className="flex flex-wrap items-center justify-between gap-4">
        <div className="flex items-center gap-2">
          {searchable && (
            <div className="relative">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
              <Input
                placeholder="Search..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="pl-9 w-64"
              />
              {hasActiveFilters && (
                <button
                  onClick={clearAllFilters}
                  className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
                >
                  <X className="w-4 h-4" />
                </button>
              )}
            </div>
          )}
          {hasActiveFilters && (
            <span className="text-sm text-gray-500">{filteredData.length} results</span>
          )}
        </div>

        <div className="flex items-center gap-2">
          {/* View toggle */}
          <div className="flex border rounded-lg overflow-hidden">
            <button
              onClick={() => setViewMode('table')}
              className={`p-2 ${viewMode === 'table' ? 'bg-green-100 text-green-700' : 'bg-white text-gray-600 hover:bg-gray-50'}`}
              title="Table view"
            >
              <List className="w-4 h-4" />
            </button>
            <button
              onClick={() => setViewMode('grid')}
              className={`p-2 ${viewMode === 'grid' ? 'bg-green-100 text-green-700' : 'bg-white text-gray-600 hover:bg-gray-50'}`}
              title="Grid view"
            >
              <Grid className="w-4 h-4" />
            </button>
          </div>

          {/* Export buttons */}
          {exportable && (
            <>
              <Button variant="outline" size="sm" onClick={handleExportCSV}>
                <Download className="w-4 h-4 mr-1" /> CSV
              </Button>
              <Button variant="outline" size="sm" onClick={handleExportPDF}>
                <Printer className="w-4 h-4 mr-1" /> PDF
              </Button>
              <Button variant="outline" size="sm" onClick={handlePrint}>
                <Printer className="w-4 h-4 mr-1" /> Print
              </Button>
            </>
          )}
        </div>
      </div>

      {/* Column Filters */}
      {columns.some(col => col.filterable) && (
        <div className="flex flex-wrap gap-2">
          {columns.filter(col => col.filterable).map(col => (
            <Input
              key={col.key}
              placeholder={`Filter ${col.header}`}
              value={columnFilters[col.key] || ''}
              onChange={(e) => setColumnFilters(prev => ({ ...prev, [col.key]: e.target.value }))}
              className="w-40 text-sm"
            />
          ))}
        </div>
      )}

      {/* Table/Grid View */}
      {filteredData.length === 0 ? (
        <Card padding="lg" className="text-center py-16">
          <div className="text-5xl mb-4">{emptyState.icon}</div>
          <h3 className="text-lg font-bold text-gray-800">{emptyState.title}</h3>
          <p className="text-gray-500">{emptyState.description}</p>
        </Card>
      ) : viewMode === 'table' ? (
        <div className="overflow-x-auto border border-gray-200 rounded-xl" ref={tableRef}>
          <table className="w-full">
            <thead className="bg-gray-50 border-b border-gray-200">
              <tr>
                {displayColumns.map(col => (
                  <th key={col.key} className="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">
                    {col.header}
                  </th>
                ))}
                {actions && <th className="px-6 py-3 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Actions</th>}
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-200">
              {paginatedData.map((item, idx) => (
                <tr
                  key={item.id || idx}
                  className={`hover:bg-gray-50 ${onRowClick ? 'cursor-pointer' : ''}`}
                  onClick={() => onRowClick?.(item)}
                >
                  {displayColumns.map(col => (
                    <td key={col.key} className="px-6 py-4 whitespace-nowrap text-sm text-gray-700">
                      {col.render ? col.render(item, idx) : (col.accessor ? col.accessor(item, idx) : item[col.key])}
                    </td>
                  ))}
                  {actions && (
                    <td className="px-6 py-4 whitespace-nowrap text-right text-sm">
                      {actions(item)}
                    </td>
                  )}
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      ) : (
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
          {paginatedData.map((item, idx) => (
            <Card key={item.id || idx} padding="md" hover className="cursor-pointer" onClick={() => onRowClick?.(item)}>
              <div className="space-y-2">
                {showSerialNumber && (
                  <div className="flex justify-between">
                    <span className="text-sm text-gray-500">S/N:</span>
                    <span className="text-sm font-medium text-gray-800">{idx + 1 + (currentPage * pageSize)}</span>
                  </div>
                )}
                {columns.slice(0, 4).map(col => (
                  <div key={col.key} className="flex justify-between">
                    <span className="text-sm text-gray-500">{col.header}:</span>
                    <span className="text-sm font-medium text-gray-800">
                      {col.render ? col.render(item, idx) : (col.accessor ? col.accessor(item, idx) : item[col.key])}
                    </span>
                  </div>
                ))}
                {actions && <div className="pt-2 border-t">{actions(item)}</div>}
              </div>
            </Card>
          ))}
        </div>
      )}

      {/* Pagination */}
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-2">
          <span className="text-sm text-gray-600">Rows per page:</span>
          <Select
            value={String(pageSize)}
            onChange={(e) => setPageSize(Number(e.target.value))}
            options={pageSizeOptions.map(n => ({ value: String(n), label: String(n) }))}
            className="w-20"
          />
          <span className="text-sm text-gray-500 ml-2">
            Showing {currentPage * pageSize + 1}-{Math.min((currentPage + 1) * pageSize, filteredData.length)} of {filteredData.length}

import { useAuth } from '../context/AuthContext'

export default function ProtectedRoute({ children, roles }) {
  const { user, loading, hasAnyRole } = useAuth()

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="text-center">
          <div className="w-10 h-10 border-4 border-green-600 border-t-transparent rounded-full animate-spin mx-auto mb-4"></div>
          <p className="text-gray-600">Checking access...</p>
        </div>
      </div>
    )
  }

  if (!user) {
    return <Navigate to="/login" replace />
  }

  if (roles && !hasAnyRole(roles)) {
    return <Navigate to="/dashboard" replace />
  }

  return children
}

import { createContext, useContext, useState, useEffect } from 'react'
import axios from 'axios'
import toast from 'react-hot-toast'

const AuthContext = createContext()
export const useAuth = () => useContext(AuthContext)

const api = axios.create({ baseURL: 'http://localhost:5000' })

api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token')
    if (token) config.headers.Authorization = `Bearer ${token}`
    return config
  },
  (error) => Promise.reject(error)
)

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    const token = localStorage.getItem('token')
    if (token) {
      api.get('/auth/me')
        .then(res => setUser(res.data))
        .catch(() => localStorage.removeItem('token'))
        .finally(() => setLoading(false))
    } else {
      setLoading(false)
    }
  }, [])

  const login = async (email, password) => {
    try {
      const res = await api.post('/auth/login', { email, password })
      localStorage.setItem('token', res.data.access_token)
      setUser(res.data.user)
      toast.success('Login successful')
      return true
    } catch (err) {
      toast.error(err.response?.data?.error || 'Login failed')
      return false
    }
  }

  const register = async (userData) => {
    try {
      const res = await api.post('/auth/register', userData)
      localStorage.setItem('token', res.data.access_token)
      setUser(res.data.user)
      toast.success('Account created')
      return true
    } catch (err) {
      toast.error(err.response?.data?.error || 'Registration failed')
      return false
    }
  }

  const logout = () => {
    localStorage.removeItem('token')
    setUser(null)
    toast.success('Logged out')
  }

  const hasRole = (role) => user?.roles?.includes(role) || false
  const hasAnyRole = (roles) => roles.some(r => user?.roles?.includes(r))

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-green-50">
        <div className="text-center">
          <div className="w-10 h-10 border-4 border-green-600 border-t-transparent rounded-full animate-spin mx-auto mb-3" />
          <p className="text-gray-600">Loading session...</p>
        </div>
      </div>
    )
  }

  return (
    <AuthContext.Provider value={{ user, loading, login, register, logout, hasRole, hasAnyRole, api }}>
      {children}
    </AuthContext.Provider>
  )
}

import { createContext, useContext, useState, useEffect } from 'react'

const CartContext = createContext()
export const useCart = () => useContext(CartContext)

export const CartProvider = ({ children }) => {
  const [items, setItems] = useState(() => {
    try {
      const saved = localStorage.getItem('cart')
      return saved ? JSON.parse(saved) : []
    } catch {
      return []
    }
  })

  useEffect(() => {
    localStorage.setItem('cart', JSON.stringify(items))
  }, [items])

  const addItem = (newItem) => {
    setItems(prev => {
      const existing = prev.find(i => i.ticket_type_id === newItem.ticket_type_id)
      if (existing) {
        return prev.map(i =>
          i.ticket_type_id === newItem.ticket_type_id
            ? {
                ...i,
                quantity: i.quantity + newItem.quantity,
                holdToken: newItem.holdToken || i.holdToken,
              }
            : i
        )
      }
      return [...prev, newItem]
    })
  }

  const removeItem = (ticketTypeId) => {
    setItems(prev => prev.filter(i => i.ticket_type_id !== ticketTypeId))
  }

  const updateQuantity = (ticketTypeId, quantity) => {
    if (quantity <= 0) {
      removeItem(ticketTypeId)
    } else {
      setItems(prev =>
        prev.map(i =>
          i.ticket_type_id === ticketTypeId ? { ...i, quantity } : i
        )
      )
    }
  }

  const clearCart = () => {
    setItems([])
    localStorage.removeItem('cart')
  }

  const totalAmount = items.reduce((sum, item) => sum + item.price * item.quantity, 0)
  const totalItems = items.reduce((sum, item) => sum + item.quantity, 0)

  return (
    <CartContext.Provider
      value={{ items, addItem, removeItem, updateQuantity, clearCart, totalAmount, totalItems }}
    >
      {children}
    </CartContext.Provider>
  )
}

import { useState, useEffect } from 'react'
import { useNavigate } from 'react-router-dom'
import { useCart } from '../context/CartContext'
import { useAuth } from '../context/AuthContext'
import MainLayout from '../layouts/MainLayout'
import { Button, Card, Input, Alert, Spinner } from '../components/ui'
import toast from 'react-hot-toast'

export default function CheckoutPage() {
  const navigate = useNavigate()
  const { items, totalAmount, updateQuantity, removeItem, clearCart } = useCart()
  const { user, api } = useAuth()
  const [loading, setLoading] = useState(false)
  const [cancelLoading, setCancelLoading] = useState(false)
  const [validating, setValidating] = useState(true)   // validating holds on load
  const [error, setError] = useState('')
  const [formData, setFormData] = useState({
    customer_name: user ? `${user.first_name || ''} ${user.last_name || ''}`.trim() : '',
    customer_email: user?.email || '',
    customer_phone: user?.phone || ''
  })

  // Re‑validate each item’s hold when the checkout page first loads
  useEffect(() => {
    if (items.length === 0) {
      setValidating(false)
      return
    }

    const validateHolds = async () => {
      for (const item of items) {
        if (!item.holdToken) continue

        try {
          // Try to refresh the hold with the same quantity
          const payload = {
            ticket_type_id: item.ticket_type_id,
            quantity: item.quantity,
            hold_token: item.holdToken,
          }
          const res = await api.post(`/api/matches/${item.match_id}/hold`, payload)
          const data = res.data

          // Update hold token if it changed
          if (data.hold_token && data.hold_token !== item.holdToken) {
            item.holdToken = data.hold_token
            updateQuantity(item.ticket_type_id, item.quantity) // trigger save
          }
        } catch (err) {
          if (err.response?.status === 409) {
            // Hold conflict – try to re‑hold with available quantity
            const available = err.response.data?.available || 0
            if (available > 0) {
              try {
                const res = await api.post(`/api/matches/${item.match_id}/hold`, {
                  ticket_type_id: item.ticket_type_id,
                  quantity: available,
                  hold_token: item.holdToken || undefined,
                })
                const data = res.data
                if (data.hold_token) item.holdToken = data.hold_token
                updateQuantity(item.ticket_type_id, available)
                toast.warning(`${item.ticket_type_name}: only ${available} seat(s) left. Your quantity has been adjusted.`)
              } catch {
                // Could not even hold the available seats, remove the item
                removeItem(item.ticket_type_id)
                toast.error(`${item.ticket_type_name} is no longer available and has been removed.`)
              }
            } else {
              // No seats left – remove the item
              removeItem(item.ticket_type_id)
              toast.error(`${item.ticket_type_name} is sold out and has been removed.`)
            }
          }
        }
      }
      setValidating(false)
    }

    validateHolds()
  }, []) // run once on mount

  if (validating) {
    return (
      <MainLayout>
        <div className="flex justify-center py-20">
          <Spinner size="xl" />
          <p className="ml-4 text-gray-500">Checking seat availability...</p>
        </div>
      </MainLayout>
    )
  }

  if (items.length === 0) {
    return (
      <MainLayout>
        <div className="container mx-auto px-4 py-16 text-center">
          <Card padding="xl" className="max-w-md mx-auto">
            <div className="text-6xl mb-4">🛒</div>
            <h2 className="text-2xl font-bold text-gray-800 mb-2">Your cart is empty</h2>
            <p className="text-gray-500 mb-6">Add some tickets to get started!</p>
            <Button to="/matches" variant="primary" size="lg">
              Browse Matches
            </Button>
          </Card>
        </div>
      </MainLayout>
    )
  }

  const handleQuantityChange = async (item, delta) => {
    const newQty = item.quantity + delta
    if (newQty < 0) return

    // Check availability via hold system
    try {
      const payload = {
        ticket_type_id: item.ticket_type_id,
        quantity: newQty,
        hold_token: item.holdToken || undefined,
      }
      const res = await api.post(`/api/matches/${item.match_id}/hold`, payload)
      const data = res.data

      if (data.hold_token) item.holdToken = data.hold_token
      updateQuantity(item.ticket_type_id, newQty)
      if (newQty === 0) toast.success('Item removed')
    } catch (err) {
      if (err.response?.status === 409) {
        const available = err.response.data?.available || 0
        if (available > 0 && available < newQty) {
          // Adjust to max available
          try {
            const res = await api.post(`/api/matches/${item.match_id}/hold`, {
              ticket_type_id: item.ticket_type_id,
              quantity: available,
              hold_token: item.holdToken || undefined,
            })

          } catch {}
        } else if (available <= 0) {
          removeItem(item.ticket_type_id)
          toast.error('Seats no longer available. Item removed.')
        }
      } else {
        toast.error('Failed to update reservation')
      }
    }
  }

  const handleRemoveItem = (item) => {
    if (item.holdToken) {
      api.delete(`/api/matches/${item.match_id}/hold/${item.holdToken}`).catch(() => {})
    }
    removeItem(item.ticket_type_id)
    toast.success('Item removed')
  }

  const handleCancelOrder = async () => {
    if (items.length === 0) return
    setCancelLoading(true)
    try {
      for (const item of items) {
        if (item.holdToken) {
          await api.delete(`/api/matches/${item.match_id}/hold/${item.holdToken}`).catch(() => {})
        }
      }
      clearCart()

    }

        items: items.map(item => ({
          ticket_type_id: item.ticket_type_id,
          quantity: item.quantity,
        })),
        hold_tokens: items.map(item => item.holdToken).filter(Boolean),
      }

      const response = await api.post('/api/orders', orderData)
      clearCart()
      toast.success('Order placed successfully!')
      navigate(`/payment/${response.data.id}`)
    } catch (error) {
      setError(error.response?.data?.error || 'Failed to place order')
    } finally {

  }

  return (

import { useEffect, useState, useRef, useCallback } from 'react'
import { useParams, useNavigate } from 'react-router-dom'
import { useAuth } from '../context/AuthContext'
import { useCart } from '../context/CartContext'
import MainLayout from '../layouts/MainLayout'
import { Button, Card, Badge, Spinner, Alert } from '../components/ui'
import toast from 'react-hot-toast'

export default function MatchDetailPage() {
  const { id } = useParams()
  const navigate = useNavigate()
  const { api } = useAuth()
  const { addItem, items: cartItems } = useCart()   // <-- need cart items
  const [match, setMatch] = useState(null)
  const [loading, setLoading] = useState(true)
  const [quantities, setQuantities] = useState({})       // local UI quantities (my hold)
  const [holdTokens, setHoldTokens] = useState({})       // token per ticket type
  const [error, setError] = useState('')
  const pollingRef = useRef(null)

  const fetchMatch = useCallback(async () => {
    try {
      const res = await api.get(`/api/matches/${id}`)
      setMatch(res.data)
    } catch (err) {
      setError('Failed to refresh match details')
    }
  }, [id, api])

  useEffect(() => {
    fetchMatch().then(() => setLoading(false))
  }, [fetchMatch])

  // Periodic refresh every 5 seconds
 /* useEffect(() => {
    pollingRef.current = setInterval(fetchMatch, 5000)
    return () => clearInterval(pollingRef.current)
  }, [fetchMatch])
*/

useEffect(() => {
  const handleVisibility = () => {
    if (document.hidden) {
      // Tab not visible – poll slowly
      clearInterval(pollingRef.current);
      pollingRef.current = setInterval(fetchMatch, 5000);
    } else {
      // Tab visible – normal interval
      clearInterval(pollingRef.current);
      pollingRef.current = setInterval(fetchMatch, 3000);
    }
  };
  document.addEventListener('visibilitychange', handleVisibility);
  return () => document.removeEventListener('visibilitychange', handleVisibility);
}, []);
  // Release only non‑cart holds on page unload
  useEffect(() => {
    const releaseAll = () => {
      Object.entries(holdTokens).forEach(([ticketTypeId, token]) => {
        // Only release if this hold token is NOT in the cart
        const isInCart = cartItems.some(item => item.holdToken === token)
        if (token && !isInCart) {
          api.delete(`/api/matches/${id}/hold/${token}`).catch(() => {})
        }
      })
    }
    window.addEventListener('beforeunload', releaseAll)
    return () => {
      window.removeEventListener('beforeunload', releaseAll)
      releaseAll()
    }
  }, [holdTokens, id, api, cartItems])

  const syncHold = async (ticketTypeId, newQuantity) => {
    const ticketType = match?.ticket_types?.find(tt => tt.id === ticketTypeId)
    if (!ticketType) return

    const currentQty = quantities[ticketTypeId] || 0
    const currentToken = holdTokens[ticketTypeId]

    if (newQuantity === currentQty && currentToken) return

    try {
      const payload = {
        ticket_type_id: ticketTypeId,
        quantity: newQuantity,
        hold_token: currentToken || undefined
      }

      const res = await api.post(`/api/matches/${id}/hold`, payload)
      const data = res.data

      if (data.hold_token && data.hold_token !== currentToken) {
        setHoldTokens(prev => ({ ...prev, [ticketTypeId]: data.hold_token }))
      }

      setQuantities(prev => ({ ...prev, [ticketTypeId]: newQuantity }))

      if (newQuantity === 0 && currentToken) {
        // Release if not in cart
        const isInCart = cartItems.some(item => item.holdToken === currentToken)
        if (!isInCart) {
          await api.delete(`/api/matches/${id}/hold/${currentToken}`)
        }
        setHoldTokens(prev => ({ ...prev, [ticketTypeId]: null }))
      }
    } catch (err) {
      if (err.response?.status === 409) {
        const available = err.response.data?.available || 0
        setQuantities(prev => ({ ...prev, [ticketTypeId]: Math.min(prev[ticketTypeId] || 0, available) }))
        toast.error('Not enough seats available. Quantity adjusted.')
        if (currentToken) {
          api.delete(`/api/matches/${id}/hold/${currentToken}`).catch(() => {})
          setHoldTokens(prev => ({ ...prev, [ticketTypeId]: null }))
        }
      } else {
        toast.error('Failed to reserve seats')
      }
    }
  }

  const handleQuantityChange = (ticketTypeId, newValue) => {
    const ticketType = match?.ticket_types?.find(tt => tt.id === ticketTypeId)
    if (!ticketType) return

    const myHoldQty = quantities[ticketTypeId] || 0
    const otherHolds = Math.max(0, (ticketType.held || 0) - myHoldQty)
    const maxForMe = ticketType.capacity - ticketType.tickets_sold - otherHolds

    let qty = parseInt(newValue, 10)
    if (isNaN(qty)) qty = 0
    qty = Math.max(0, Math.min(qty, maxForMe))

    setQuantities(prev => ({ ...prev, [ticketTypeId]: qty }))
    syncHold(ticketTypeId, qty)
  }

  const handleIncrement = (ticketTypeId) => {
    const ticketType = match?.ticket_types?.find(tt => tt.id === ticketTypeId)
    if (!ticketType) return

    const myHoldQty = quantities[ticketTypeId] || 0
    const otherHolds = Math.max(0, (ticketType.held || 0) - myHoldQty)
    const maxForMe = ticketType.capacity - ticketType.tickets_sold - otherHolds

    if (myHoldQty < maxForMe) {
      handleQuantityChange(ticketTypeId, myHoldQty + 1)
    }
  }

  const handleDecrement = (ticketTypeId) => {
    const current = quantities[ticketTypeId] || 0
    if (current > 0) {
      handleQuantityChange(ticketTypeId, current - 1)
    }
  }

  const handleAddToCart = () => {
    let added = false
    match.ticket_types.forEach(tt => {
      const qty = quantities[tt.id] || 0
      if (qty > 0) {
        addItem({
          ticket_type_id: tt.id,
          ticket_type_name: tt.name,
          match_id: match.id,
          match_name: `${match.home_team} vs ${match.away_team}`,
          price: tt.price,
          quantity: qty,
          holdToken: holdTokens[tt.id]     // store the hold token
        })
        added = true
      }
    })
    if (added) toast.success('Added to cart!')
  }

  const handleBuyNow = () => {
    handleAddToCart()
    navigate('/checkout')
  }

  if (loading) {
    return (
      <MainLayout>
        <div className="flex justify-center py-20"><Spinner size="xl" /></div>
      </MainLayout>
    )
  }

  if (error || !match) {
    return (
      <MainLayout>
        <div className="container mx-auto px-4 py-8">
          <Alert variant="error">{error || 'Match not found'}</Alert>
        </div>
      </MainLayout>
    )
  }

  const isNational = match.match_type === 'national'
  const totalItems = Object.values(quantities).reduce((a, b) => a + b, 0)

  return (
    <MainLayout>
      <div className="container mx-auto px-4 py-8">
        <div className="max-w-4xl mx-auto">
          {/* Match Header */}
          <div className={`rounded-2xl p-8 text-white mb-8 ${isNational ? 'bg-gradient-to-r from-green-600 to-green-700' : 'bg-gradient-to-r from-orange-500 to-orange-600'}`}>
            <Badge variant={isNational ? 'success' : 'orange'} className="mb-4">
              {isNational ? '🇿🇲 National Team' : '🏆 Club Match'}
            </Badge>
            <h1 className="text-3xl md:text-4xl font-bold mb-3">
              {match.home_team} vs {match.away_team}
            </h1>
            <div className="flex flex-wrap gap-6 text-white/90">
              <span className="flex items-center gap-2">📍 {match.venue}</span>
              <span className="flex items-center gap-2">📅 {new Date(match.date_time).toLocaleString()}</span>
            </div>
            <div className="mt-6 grid grid-cols-3 gap-4">
              <div className="bg-white/20 backdrop-blur-sm rounded-xl p-3 text-center">
                <p className="text-2xl font-bold">{match.total_capacity ?? '—'}</p>
                <p className="text-xs opacity-80">Total Seats</p>

              <div className="bg-white/20 backdrop-blur-sm rounded-xl p-3 text-center">
                <p className="text-2xl font-bold">{match.total_tickets_sold ?? '—'}</p>
                <p className="text-xs opacity-80">Sold</p>

                <p className="text-xs opacity-80">Available</p>

            </div>
          </div>

          <Card padding="lg">
            <h2 className="text-xl font-bold text-gray-800 mb-6">Select Your Tickets</h2>
            
            {match.ticket_types?.length === 0 ? (
              <Alert variant="warning">No ticket types available for this match.</Alert>
            ) : (
              <div className="space-y-4">
                {match.ticket_types.map(tt => {
                  const myHoldQty = quantities[tt.id] || 0
                  const otherHolds = Math.max(0, (tt.held || 0) - myHoldQty)
                  const maxForMe = tt.capacity - tt.tickets_sold - otherHolds
                  const freeForMe = maxForMe - myHoldQty
                  const showControls = myHoldQty > 0 || maxForMe > 0

const sanitizeErrorMessage = (message) => {
  if (!message) return ''
  let sanitized = message
  for (const [code, name] of Object.entries(providerNames)) {
    sanitized = sanitized.replace(new RegExp(code, 'g'), name)
    const shortCode = code.replace('_ZMB', '').replace('_OAPI_ZMB', '')
    if (shortCode !== code) {
      sanitized = sanitized.replace(new RegExp(shortCode, 'g'), name)
    }
  }
  return sanitized
}

    fetchProviders()
    fetchOrder()
    return () => {
      if (pollingRef.current) clearInterval(pollingRef.current)
    }

      setProviders(res.data)

      toast.error('Failed to load payment providers')
    }
  }

  const fetchOrder = async () => {

      const res = await api.get(`/api/orders/${orderId}`)
      if (res.data.payment_status !== 'pending') {

        navigate('/checkout')
        return
      }

    } catch (err) {

      navigate('/checkout')

      setLoading(false)
    }

      setCancelling(false)
    }

    let input = e.target.value.replace(/\D/g, '')
    if (input.length > 10) input = input.slice(0, 10)
    if (input.length > 0 && input[0] !== '0') return
    setPhoneNumber(input)

          else if (code === 'MISSING_PARAMETER') setError('Missing payment details.')
          else if (code === 'INVALID_PARAMETER') setError('Invalid payment parameter.')
          else if (code === 'UNSUPPORTED_PARAMETER') setError('Unsupported payment parameter.')
          else if (code === 'AUTHENTICATION_ERROR' || code === 'AUTHORISATION_ERROR') setError('Payment service authentication error. Contact support.')
          else if (code === 'UNKNOWN_ERROR') setError('Payment service error. Try again later.')

import { useState } from 'react'
import axios from 'axios'
import toast from 'react-hot-toast'
import { useAuth } from '../context/AuthContext'

export default function VerifyTicketPage() {
  const { api } = useAuth()
  const [ticketNumber, setTicketNumber] = useState('')
  const [verifying, setVerifying] = useState(false)
  const [result, setResult] = useState(null)

  const handleVerify = async (e) => {
    e.preventDefault()
    if (!ticketNumber.trim()) return

    setVerifying(true)
    setResult(null)
    try {
      const response = await api.post('/api/tickets/verify', { ticket_number: ticketNumber })
      setResult({ success: true, data: response.data })
      toast.success('Ticket verified successfully!')
    } catch (error) {
      setResult({ success: false, message: error.response?.data?.message || 'Verification failed' })
      toast.error(error.response?.data?.message || 'Invalid ticket')
    } finally {
      setVerifying(false)
    }
  }

  return (
    <div className="max-w-2xl mx-auto">
      <h1 className="text-3xl font-bold mb-6">Verify Ticket</h1>
      
      <div className="bg-white rounded-lg shadow p-6">
        <form onSubmit={handleVerify} className="flex gap-4 mb-6">
          <input
            type="text"
            value={ticketNumber}
            onChange={(e) => setTicketNumber(e.target.value)}
            placeholder="Enter ticket number or scan QR code"
            className="flex-1 border rounded px-4 py-2 text-lg"
            autoFocus
          />
          <button
            type="submit"
            disabled={verifying || !ticketNumber.trim()}
            className="bg-green-600 text-white px-6 py-2 rounded hover:bg-green-700 disabled:opacity-50"
          >
            {verifying ? 'Verifying...' : 'Verify'}
          </button>
        </form>

        {result && (
          <div className={`border rounded-lg p-4 ${result.success ? 'bg-green-50 border-green-300' : 'bg-red-50 border-red-300'}`}>
            {result.success ? (
              <div>

              </div>
            ) : (

      resetForm()
      fetchUsers()
    } catch (error) {

    e.preventDefault()

    try {
      const payload = { ...formData }

function ReportsManager() {
  const { api } = useAuth()
  const [stats, setStats] = useState({ totalRevenue: 0, totalOrders: 0, totalTickets: 0, matchesSold: [] })
  const [loading, setLoading] = useState(true)

  useEffect(() => { fetchReportData() }, [])

  const fetchReportData = async () => {
    try {
      const [ordersRes, matchesRes] = await Promise.all([api.get('/api/admin/orders'), api.get('/api/matches?active_only=false')])
      const orders = ordersRes.data
      const totalRevenue = orders.reduce((sum, o) => sum + parseFloat(o.total_amount), 0)
      const totalTickets = orders.flatMap(o => o.items || []).reduce((sum, item) => sum + item.quantity, 0)
      const matchSales = {}
      orders.forEach(order => { order.items?.forEach(item => { const matchId = item.match_id; if (!matchSales[matchId]) matchSales[matchId] = { matchName: item.match_name, tickets: 0, revenue: 0 }; matchSales[matchId].tickets += item.quantity; matchSales[matchId].revenue += item.quantity * item.unit_price }) })
      const matchesSold = Object.entries(matchSales).map(([id, data]) => ({ id, ...data })).sort((a, b) => b.revenue - a.revenue).slice(0, 10)
      setStats({ totalRevenue, totalOrders: orders.length, totalTickets, matchesSold })
    } catch (error) { toast.error('Failed to load reports') } finally { setLoading(false) }
  }

  if (loading) return <div className="flex justify-center py-20"><Spinner size="xl" /></div>

  return (
    <div className="space-y-6">
      <div><h1 className="text-2xl font-bold text-gray-800 mb-1">Reports & Analytics</h1><p className="text-gray-500">Sales and performance overview</p></div>
      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
        <StatCard title="Total Revenue" value={`ZMW ${stats.totalRevenue.toFixed(2)}`} icon="💰" color="green" />
        <StatCard title="Total Orders" value={stats.totalOrders} icon="📦" color="blue" />
        <StatCard title="Tickets Sold" value={stats.totalTickets} icon="🎫" color="orange" />
      </div>
      <Card padding="lg"><h3 className="font-bold text-gray-800 mb-4">Top Selling Matches</h3>
        {stats.matchesSold.length > 0 ? (
          <table className="w-full"><thead><tr className="border-b"><th className="text-left py-2">Match</th><th className="text-left">Tickets</th><th className="text-left">Revenue</th></tr></thead>
            <tbody>{stats.matchesSold.map(m => (<tr key={m.id} className="border-b"><td className="py-2">{m.matchName}</td><td>{m.tickets}</td><td>ZMW {m.revenue.toFixed(2)}</td></tr>))}</tbody>
          </table>
        ) : <EmptyState icon="📊" title="No sales data yet" description="Sales will appear here once tickets are purchased." />}
      </Card>
      <Card padding="lg"><h3 className="font-bold text-gray-800 mb-4">Quick Insights</h3>
        <div className="grid grid-cols-2 gap-4">
          <div className="p-4 bg-green-50 rounded-xl"><p className="text-sm text-gray-500">Average Order Value</p><p className="text-2xl font-bold">ZMW {stats.totalOrders ? (stats.totalRevenue / stats.totalOrders).toFixed(2) : '0.00'}</p></div>
          <div className="p-4 bg-blue-50 rounded-xl"><p className="text-sm text-gray-500">Tickets per Order</p><p className="text-2xl font-bold">{stats.totalOrders ? (stats.totalTickets / stats.totalOrders).toFixed(1) : '0'}</p></div>
        </div>
      </Card>
    </div>
  )
}

  const [submitting, setSubmitting] = useState(false)

  useEffect(() => {
    if ((user.roles.includes('club_admin') || user.roles.includes('club_officer')) && !user.club_id) {
      setNoClubAssigned(true)
      setLoading(false)
      return
    }
    fetchMatches()
  }, [user])

  const fetchMatches = async () => {
    try {
      const res = await api.get('/api/club/matches')   // ← Changed
      setMatches(res.data)
    } catch (error) {
      toast.error('Failed to load matches')
    } finally {
      setLoading(false)
    }
  }

  const addTicketType = () => {
    setFormData({
      ...formData,
      ticket_types: [...formData.ticket_types, { name: '', price: '', capacity: '' }]
    })
  }

  const updateTicketType = (index, field, value) => {
    const updated = [...formData.ticket_types]
    updated[index][field] = value
    setFormData({ ...formData, ticket_types: updated })
  }

  const removeTicketType = (index) => {
    if (formData.ticket_types.length > 1) {
      setFormData({
        ...formData,
        ticket_types: formData.ticket_types.filter((_, i) => i !== index)
      })
    }
  }

  const handleSubmit = async (e) => {
    e.preventDefault()
    setSubmitting(true)
    try {
      const submitData = {
        ...formData,
        ticket_types: formData.ticket_types.map(tt => ({
          ...tt,
          price: parseFloat(tt.price),
          capacity: parseInt(tt.capacity)
        }))
      }
      await api.post('/api/club/matches', submitData)   // ← Changed
      toast.success('Match created successfully!')
      setShowForm(false)
      setFormData({
        home_team: '',
        away_team: '',
        venue: '',
        date_time: '',
        description: '',
        ticket_types: [{ name: 'Regular', price: '', capacity: '' }]
      })
      fetchMatches()
    } catch (error) {
      toast.error('Failed to create match')
    } finally {
      setSubmitting(false)
    }
  }

  if (noClubAssigned) {
    return (
      <div className="space-y-6">
        <div>
          <h1 className="text-2xl font-bold text-gray-800 mb-1">Club Matches</h1>
          <p className="text-gray-500">Manage your club's fixtures</p>
        </div>
        <Card padding="lg">
          <div className="text-center py-8">
            <div className="text-6xl mb-4">⚽</div>
            <h3 className="text-xl font-bold text-gray-800 mb-2">No Club Assigned</h3>
            <p className="text-gray-600 max-w-md mx-auto">
              Your account is not linked to a club. Please contact an administrator to get set up.
            </p>
          </div>
        </Card>
      </div>
    )
  }

  if (loading) {
    return (
      <div className="flex justify-center py-20">
        <Spinner size="xl" />
      </div>
    )
  }

  return (
    <div className="space-y-6">
      <div className="flex justify-between items-center">
        <div>
          <h1 className="text-2xl font-bold text-gray-800 mb-1">Club Matches</h1>
          <p className="text-gray-500">Manage your club's fixtures</p>
        </div>
        <Button onClick={() => setShowForm(!showForm)} variant={showForm ? 'outline' : 'primary'}>
          {showForm ? 'Cancel' : '+ New Match'}
        </Button>
      </div>

      {showForm && (
        <Card padding="lg">
          <h3 className="font-bold text-gray-800 mb-4">Create New Match</h3>
          <form onSubmit={handleSubmit} className="space-y-4">
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <Input
                label="Home Team"
                placeholder="e.g., ZESCO United"
                value={formData.home_team}
                onChange={e => setFormData({ ...formData, home_team: e.target.value })}
                required
              />
              <Input
                label="Away Team"
                placeholder="e.g., Nkana FC"
                value={formData.away_team}
                onChange={e => setFormData({ ...formData, away_team: e.target.value })}
                required
              />
              <Input
                label="Venue"
                placeholder="e.g., Levy Mwanawasa Stadium"
                value={formData.venue}
                onChange={e => setFormData({ ...formData, venue: e.target.value })}
                required
              />
              <Input
                label="Date & Time"
                type="datetime-local"
                value={formData.date_time}
                onChange={e => setFormData({ ...formData, date_time: e.target.value })}
                required
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">Ticket Types</label>
              <div className="space-y-3">
                {formData.ticket_types.map((tt, idx) => (
                  <div key={idx} className="flex items-center gap-2">
                    <div className="grid grid-cols-3 gap-2 flex-1">
                      <Input
                        placeholder="Name (e.g., VIP)"
                        value={tt.name}
                        onChange={e => updateTicketType(idx, 'name', e.target.value)}
                        required
                      />
                      <Input
                        type="number"
                        placeholder="Price (ZMW)"
                        value={tt.price}
                        onChange={e => updateTicketType(idx, 'price', e.target.value)}
                        min="0"
                        step="0.01"
                        required
                      />
                      <Input
                        type="number"
                        placeholder="Capacity"
                        value={tt.capacity}
                        onChange={e => updateTicketType(idx, 'capacity', e.target.value)}
                        min="1"
                        required
                      />
                    </div>
                    {formData.ticket_types.length > 1 && (
                      <button
                        type="button"
                        onClick={() => removeTicketType(idx)}
                        className="p-2 text-red-500 hover:bg-red-50 rounded-lg"
                      >
                        🗑️
                      </button>
                    )}
                  </div>
                ))}
              </div>
              <button
                type="button"
                onClick={addTicketType}
                className="mt-3 text-sm text-orange-600 hover:underline"
              >
                + Add Another Ticket Type
              </button>
            </div>

            <div className="flex gap-3 pt-4">
              <Button type="submit" loading={submitting}>
                Create Match
              </Button>
              <Button type="button" variant="outline" onClick={() => setShowForm(false)}>
                Cancel
              </Button>
            </div>
          </form>
        </Card>
      )}

      <Card padding="none">
        {matches.length === 0 ? (
          <EmptyState
            icon="⚽"
            title="No matches found"
            description="Create your first match to start selling tickets."
            actionLabel="Create Match"
            onAction={() => setShowForm(true)}
          />
        ) : (
          <Table>
            <TableHead>
              <TableRow>
                <TableHeaderCell>Match</TableHeaderCell>
                <TableHeaderCell>Venue</TableHeaderCell>
                <TableHeaderCell>Date & Time</TableHeaderCell>
                <TableHeaderCell>Status</TableHeaderCell>
                <TableHeaderCell align="right">Actions</TableHeaderCell>
              </TableRow>
            </TableHead>
            <TableBody>
              {matches.map(match => (
                <TableRow key={match.id}>
                  <TableCell className="font-medium">
                    {match.home_team} vs {match.away_team}
                  </TableCell>
                  <TableCell>{match.venue}</TableCell>
                  <TableCell>{new Date(match.date_time).toLocaleString()}</TableCell>
                  <TableCell>
                    <Badge variant={match.is_active ? 'success' : 'secondary'}>
                      {match.is_active ? 'Active' : 'Inactive'}
                    </Badge>
                  </TableCell>
                  <TableCell align="right">
                    <div className="flex justify-end gap-2">
                      <button className="text-blue-600 hover:underline text-sm">Edit</button>
                      <button className="text-red-600 hover:underline text-sm">Cancel</button>
                    </div>
                  </TableCell>
                </TableRow>
              ))}
            </TableBody>
          </Table>
        )}
      </Card>
    </div>
  )
}

// -------------------- ClubTickets --------------------
function ClubTickets() {
  const { api } = useAuth()
  const [ticketSales, setTicketSales] = useState([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    const fetchSales = async () => {
      try {
        // const res = await api.get('/api/club/tickets/sales')
        // setTicketSales(res.data)
        setTicketSales([])
      } catch (error) {
        toast.error('Failed to load ticket sales')
      } finally {
        setLoading(false)
      }
    }
    fetchSales()
  }, [api])

  if (loading) {
    return (
      <div className="flex justify-center py-20">
        <Spinner size="xl" />
      </div>
    )
  }

  return (
    <div className="space-y-6">
      <div>
        <h1 className="text-2xl font-bold text-gray-800 mb-1">Ticket Sales</h1>
        <p className="text-gray-500">View ticket sales for your matches</p>
      </div>

      <Card padding="none">
        {ticketSales.length === 0 ? (
          <EmptyState
            icon="🎫"
            title="No ticket sales yet"
            description="Sales data will appear here once tickets are purchased."
          />
        ) : (
          <Table>
            <TableHead>
              <TableRow>
                <TableHeaderCell>Match</TableHeaderCell>
                <TableHeaderCell>Ticket Type</TableHeaderCell>
                <TableHeaderCell>Sold</TableHeaderCell>
                <TableHeaderCell>Revenue</TableHeaderCell>
              </TableRow>
            </TableHead>
            <TableBody>
              {ticketSales.map((sale, idx) => (
                <TableRow key={idx}>
                  <TableCell>{sale.match}</TableCell>
                  <TableCell>{sale.type}</TableCell>
                  <TableCell>{sale.sold}</TableCell>
                  <TableCell>ZMW {sale.revenue.toFixed(2)}</TableCell>
                </TableRow>
              ))}
            </TableBody>
          </Table>
        )}
      </Card>
    </div>
  )
}
